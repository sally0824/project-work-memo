  // 全周期甘特：项目总周期 + 阶段/专题 + 工作事项 + 交付节点
  function renderGantt(){
    const box=document.getElementById('ganttContainer'),project=state.project;
    const datedPhases=state.phases.filter(p=>validDate(p.startDate)&&validDate(p.endDate));
    const tasks=state.records.filter(r=>validDate(r.planStart)&&validDate(r.planEnd)).sort((a,b)=>a.planStart.localeCompare(b.planStart));
    const deliverables=state.deliverables.filter(d=>validDate(d.dueDate)).sort((a,b)=>a.dueDate.localeCompare(b.dueDate));
    const dateStrings=[];if(validDate(project.startDate))dateStrings.push(project.startDate);if(validDate(project.endDate))dateStrings.push(project.endDate);datedPhases.forEach(x=>dateStrings.push(x.startDate,x.endDate));tasks.forEach(x=>{dateStrings.push(x.planStart,x.planEnd);if(validDate(x.actualEnd))dateStrings.push(x.actualEnd)});deliverables.forEach(x=>dateStrings.push(x.dueDate));
    if(!dateStrings.length){box.innerHTML='<div class="empty-state"><strong>暂无可生成甘特图的日期数据</strong>建议先在“项目设置”维护项目计划起止日期，再维护阶段或工作事项排期。</div>';document.getElementById('ganttRangeText').textContent='';return}
    let min=new Date(Math.min(...dateStrings.map(dateObj))),max=new Date(Math.max(...dateStrings.map(dateObj)));if(min.getTime()===max.getTime()){min.setDate(min.getDate()-7);max.setDate(max.getDate()+7)}
    const totalDays=Math.max(1,Math.round((max-min)/86400000)+1),pos=d=>Math.max(0,Math.min(100,(dateObj(d)-min)/86400000/(totalDays-1||1)*100));
    const widthBetween=(a,b)=>Math.max(.5,(dateObj(b)-dateObj(a))/86400000/(totalDays-1||1)*100 + 100/(totalDays||1));
    const scale=document.getElementById('ganttScale').value;let step,labeler;if(scale==='week'||(scale==='auto'&&totalDays<=120)){step=7;labeler=d=>`${d.getMonth()+1}/${d.getDate()}`}else{step=30;labeler=d=>`${d.getFullYear()}/${String(d.getMonth()+1).padStart(2,'0')}`}
    let marks='';for(let i=0;i<totalDays;i+=step){const d=new Date(min);d.setDate(d.getDate()+i);marks+=`<span class="day-label" style="left:${i/(totalDays-1||1)*100}%;width:${Math.min(step,totalDays-i)/(totalDays||1)*100}%">${labeler(d)}</span>`}
    const now=today(),todayLine=validDate(now)&&dateObj(now)>=min&&dateObj(now)<=max?`<span class="today-line" style="left:${pos(now)}%" title="今天"></span>`:'';
    const timeline=(inner='')=>`<div class="timeline">${inner}${todayLine}</div>`;
    const row=(kind,name,meta,inner)=>`<div class="gantt-row"><div class="gantt-label"><strong><span class="kind">${esc(kind)}</span>${esc(name||'未命名')}</strong><span>${esc(meta||'')}</span></div>${timeline(inner)}</div>`;
    let body='';
    if(validDate(project.startDate)&&validDate(project.endDate)){body+=`<div class="gantt-section"><div>项目总周期</div><div></div></div>`;body+=row('项目',project.name||'项目总周期',`${fmtDate(project.startDate)} — ${fmtDate(project.endDate)}`,`<span class="bar project" style="left:${pos(project.startDate)}%;width:${widthBetween(project.startDate,project.endDate)}%"></span>`)}
    if(datedPhases.length){body+=`<div class="gantt-section"><div>阶段 / 专题</div><div></div></div>`;body+=datedPhases.map(p=>row(p.type||'阶段',p.name,`${fmtDate(p.startDate)} — ${fmtDate(p.endDate)}`,`<span class="bar phase" style="left:${pos(p.startDate)}%;width:${widthBetween(p.startDate,p.endDate)}%"></span>`)).join('')}
    if(tasks.length){body+=`<div class="gantt-section"><div>工作事项</div><div></div></div>`;body+=tasks.map(r=>{const p1=pos(r.planStart),w=widthBetween(r.planStart,r.planEnd),completion=Math.max(0,Math.min(100,(r.completion!==''&&r.completion!=null)?num(r.completion):(r.status==='已完成'?100:0))),progress=completion?`<span class="bar progress-fill" style="left:${p1}%;width:${w*completion/100}%"></span>`:'',actual=validDate(r.actualEnd)?`<span class="bar actual-mark" style="left:${pos(r.actualEnd)}%" title="实际完成 ${fmtDate(r.actualEnd)}"></span>`:'';return row('事项',r.taskName,`${r.phase||'未分阶段'} · ${r.consultingParty||'未填责任人'} · ${completion||0}%`,`<span class="bar plan" style="left:${p1}%;width:${w}%"></span>${progress}${actual}`)}).join('')}
    if(deliverables.length){body+=`<div class="gantt-section"><div>项目交付节点</div><div></div></div>`;body+=deliverables.map(d=>row('交付',d.name||d.content,`${fmtDate(d.dueDate)} · ${d.status||'未启动'}`,`<span class="bar deliverable-mark" style="left:${pos(d.dueDate)}%" title="交付节点 ${fmtDate(d.dueDate)}"></span>`)).join('')}
    box.innerHTML=`<div class="gantt" style="--days:${totalDays}"><div class="gantt-head"><div class="gantt-label"><strong>项目层级 / 事项</strong><span>红线为今天</span></div><div class="timeline head">${marks}${todayLine}</div></div>${body}</div>`;
    const projectText=validDate(project.startDate)&&validDate(project.endDate)?`项目周期 ${fmtDate(project.startDate)} — ${fmtDate(project.endDate)} · `:'';document.getElementById('ganttRangeText').textContent=`${projectText}显示范围 ${fmtDate(min.toISOString().slice(0,10))} — ${fmtDate(max.toISOString().slice(0,10))}`;
  }
  document.getElementById('ganttScale').onchange=renderGantt;
